% Slide 3: Problem Statement
\section{Problem Statement}
\begin{frame}{Problem Statement}
    \textbf{The Challenge:}
    \begin{itemize}
        \item Traditional institutions rely on paper-based records or fragmented legacy software.
        \item Admissions, course assignments, and daily attendance are tracked in silos.
        \item This leads to data redundancy, inefficient administration, and lack of real-time insights for students.
    \end{itemize}
\end{frame}

% Slide 4: Objectives
\section{Objectives}
\begin{frame}{Objectives}
    \textbf{What we set out to build:}
    \begin{itemize}
        \item A centralized, dual-role (Admin/Student) web portal.
        \item Automate the admission pipeline using asynchronous requests.
        \item Empower administrators with full CRUD control over Users, Faculty, and Courses.
        \item Provide students with real-time analytics regarding their academic attendance.
        \item Enforce enterprise-grade security (Password Hashing \& SQL Injection mitigation).
    \end{itemize}
\end{frame}

% Slide 5: Technologies Used
\section{Technologies Used}
\begin{frame}{Technologies Used}

    \vspace{0.5cm}

% Slide 6: Main Features
\section{Main Features}
\begin{frame}{Main Features}
    \begin{itemize}
        \item \textbf{Role-Based Authentication:} Strict routing preventing unauthorized access.
        \item \textbf{Dynamic Dashboards:} Live SQL count queries rendering real-time statistics.
        \item \textbf{Relational Academics:} Admins create courses, assign faculty, and enroll students.
        \item \textbf{Attendance Tracking:} Daily attendance marking mapped to specific students and courses.
        \item \textbf{Auto-Logout Security:} Idle-session termination after 90 seconds (accompanied by animated UI toasts).
    \item \textbf{Intellectual Property:} Multi-layered codebase watermarking \& Easter Eggs.`n      \end{itemize}`n  \end{frame}`n`n  % Slide 7: System Workflow (ER Diagram Concept)
\section{System Workflow}
\begin{frame}{System Workflow \& Relational Schema}
    Our database is heavily normalized using Foreign Keys and \texttt{ON DELETE CASCADE}:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{User Table:} Stores credentials (Admin/Student).
        \item \textbf{Teacher Table:} Stores faculty profiles.
        \item \textbf{Course Table:} Links a subject to a specific \texttt{teacher\_id}.
        \item \textbf{Enrollment Table:} A junction table mapping \texttt{student\_id} to \texttt{course\_id}.
        \item \textbf{Attendance Table:} Tracks daily Present/Absent status linked to the Enrollment.
    \end{itemize}
\end{frame}
