% SLIDE 2: Introduction
\begin{frame}{Introduction}
    \textbf{What is Aether Academy?}
    \vspace{0.3cm}
    \begin{itemize}
        \item A centralized, dual-role (Admin \& Student) digital platform.
        \item Replaces legacy, paper-based institutional record-keeping.
        \item Provides real-time insights into student admissions, course enrollments, and daily attendance.
        \item Engineered with modern, asynchronous web development paradigms.
    \end{itemize}
\end{frame}

% SLIDE 3: Problem Statement
\begin{frame}{Problem Statement}
    \textbf{Challenges with Traditional Administration:}
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Data Fragmentation:} Student records are scattered across manual ledgers and isolated spreadsheets.
        \item \textbf{Attendance Opacity:} Students lack real-time visibility into their own attendance percentages, risking default.
        \item \textbf{Administrative Bottlenecks:} Manual processing of admission forms is time-consuming and error-prone.
        \item \textbf{Security Vulnerabilities:} Legacy systems often lack proper session management and cryptography.
    \end{itemize}
\end{frame}

% SLIDE 4: Objectives & Scope
\begin{frame}{Objectives \& Scope}
    \textbf{Core Objectives:}
    \begin{itemize}
        \item Automate the admission pipeline using AJAX asynchronous workflows.
        \item Separate business logic into distinct Administrator and Student portals.
        \item Provide automated, algorithmic attendance calculation.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Project Scope:}
    \begin{itemize}
        \item End-to-end management of Users, Faculty, Courses, and Attendance tracking within a singular ecosystem.
    \end{itemize}
\end{frame}

% SLIDE 5: Technologies Used (Frontend)
\begin{frame}{Technologies Used (Frontend)}
    The UI is designed around a bespoke "Night Owl" aesthetic utilizing Glassmorphism.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{HTML5 \& CSS3:} Semantic markup combined with deep CSS overrides for slate/neon color gradients.
        \item \textbf{Bootstrap 5:} Mobile-first, responsive grid framework powering the dashboard cards.
        \item \textbf{JavaScript (ES6):} Client-side DOM manipulation and idle-session detection logic.
        \item \textbf{jQuery \& AJAX:} Facilitates asynchronous HTTP POST requests (Admission Forms) without page reloads.
    \end{itemize}
\end{frame}

% SLIDE 6: Technologies Used (Backend)
\begin{frame}{Technologies Used (Backend)}
    The system architecture relies on the robust LAMP stack configuration.
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{PHP 8:} Handles core server-side business logic, API endpoints, and dynamic HTML rendering.
        \item \textbf{MySQLi:} Procedural PHP extension utilized for all database transactions.
        \item \textbf{MySQL Database:} Relational Database Management System (RDBMS).
        \item \textbf{Apache:} Local web server environment provided via XAMPP.
    \end{itemize}
\end{frame}

% SLIDE 7: Database Architecture
\begin{frame}{Database Architecture}
    The backend utilizes a heavily normalized relational database schema (\texttt{collegepr}):
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{user:} Central authentication node (Stores BCrypt hashes).
        \item \textbf{teacher:} Faculty metadata node.
        \item \textbf{course:} Maps academic subjects to a specific \texttt{teacher\_id} via Foreign Key.
        \item \textbf{enrollment:} A junction table mapping Students (M) to Courses (N).
        \item \textbf{attendance:} Transactional table utilizing \texttt{UNIQUE} constraints to prevent duplicate daily records.
    \end{itemize}
\end{frame}

% SLIDE 8: Administrator Features
\begin{frame}{Administrator Features}
    The Admin Portal provides full CRUD (Create, Read, Update, Delete) superiority:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Admissions Control:} Approve or reject incoming student applications.
        \item \textbf{Academic Structuring:} Generate new courses and assign faculty.
        \item \textbf{Enrollment Engine:} Enroll registered students into specific curriculums.
        \item \textbf{Daily Attendance:} Fast, drop-down driven interface to mark Present/Absent status across entire classes dynamically.
    \end{itemize}
\end{frame}

% SLIDE 9: Student Features
\begin{frame}{Student Features}
    The Student Portal focuses on read-only analytics and aesthetic visibility:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Curriculum Grid:} Dynamically rendered, color-coded grid cards displaying all enrolled courses.
        \item \textbf{Attendance Analytics:} Algorithmic calculation dividing \textit{Classes Attended} by \textit{Total Classes Held} to generate real-time percentages.
        \item \textbf{Responsive UI:} A collapsible sidebar and dynamic "Back to Dashboard" routing for seamless navigation.
    \end{itemize}
\end{frame}

% SLIDE 10: Security Architecture
\begin{frame}{Security Architecture}
    Enterprise-grade security mechanisms actively defend the platform:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Cryptography:} All plaintext passwords are automatically encrypted via PHP's \texttt{password\_hash()} using the BCrypt algorithm.
        \item \textbf{SQL Injection Defense:} Implementation of \texttt{mysqli\_real\_escape\_string()} on all user inputs.
        \item \textbf{Idle-Session Termination:} A custom JavaScript listener monitors mouse/keyboard activity. After 90 seconds of inactivity, the session is destroyed and a UI toast alert is generated.
    \end{itemize}
\end{frame}

% SLIDE 11: Intellectual Property Watermarking
\begin{frame}{Intellectual Property Watermarking}
    A multi-layered authorship verification system was engineered to protect the codebase:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{SQL Metadata Injection:} An immutable, hidden `\_system\_authorship` database table is bundled with the SQL dump.
        \item \textbf{HTTP Headers:} Custom X-Author PHP network headers.
        \item \textbf{Terminal Signatures:} Global console logs outputting color-coded developer badges.
        \item \textbf{Easter Egg Trigger:} Bypassing the authentication gateway via a specific keyword triggers a hidden UI authorship modal.
    \end{itemize}
\end{frame}

    \end{itemize}
\end{frame}
