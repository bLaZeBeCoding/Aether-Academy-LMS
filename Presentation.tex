% SLIDE 2: Introduction & Objectives
\begin{frame}{Introduction \& Objectives}
    \textbf{Problem:} Traditional educational institutes rely on fragmented, paper-based records for managing student admissions, courses, and daily attendance. \\
    \vspace{0.5cm}
    \textbf{Solution:} Aether Academy is a centralized, digital Student Management System.
    \vspace{0.3cm}
    \begin{itemize}
        \item Establishes a secure, dual-role (Admin \& Student) portal.
        \item Automates the admission pipeline via asynchronous workflows.
        \item Replaces manual attendance ledgers with real-time analytics.
    \end{itemize}
\end{frame}

% SLIDE 3: Technologies Used
\begin{frame}{Technologies Used}
    The system is built on the robust \textbf{LAMP Stack} combined with a modern UI framework:
    \vspace{0.5cm}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Frontend:}
        \begin{itemize}
            \item \textbf{HTML5 \& CSS3:} Custom "Night Owl" Glassmorphism theme.
            \item \textbf{Bootstrap 5:} Responsive, mobile-first grid layouts.
            \item \textbf{JavaScript/jQuery:} Asynchronous form submission and dynamic UI logic.
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Backend \& Database:}
        \begin{itemize}
            \item \textbf{PHP 8:} Server-side business logic and secure session management.
            \item \textbf{MySQL:} Normalized relational database (Entities: User, Course, Enrollment, Attendance).
        \end{itemize}
    \end{columns}
\end{frame}

% SLIDE 4: System Architecture
\begin{frame}{System Architecture \& Workflow}
    The architecture is strictly role-based, ensuring data isolation:
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Administrator Workflow:}
        \begin{itemize}
            \item Full CRUD capabilities (Create/Read/Update/Delete).
            \item Manages pending admissions, faculty profiles, and course creation.
            \item Responsible for marking daily attendance.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Student Workflow:}
        \begin{itemize}
            \item Read-only access to academic data.
            \item Dynamic dashboard displaying enrolled courses and calculated attendance percentages.
        \end{itemize}
    \end{itemize}
\end{frame}

% SLIDE 5: Key Features & Security
\begin{frame}{Key Features \& Security Mechanisms}
    \begin{itemize}
        \item \textbf{Dynamic Dashboards:} Live SQL aggregate queries power clickable UI cards for instant navigation.
        \item \textbf{Advanced Attendance Tracking:} Real-time percentage calculation with upsert database logic to prevent duplicate records.
        \item \textbf{Password Cryptography:} All user passwords are encrypted using the BCrypt hashing algorithm (\texttt{password\_hash}).
        \item \textbf{Auto-Logout Security:} A custom JavaScript timer terminates active PHP sessions after 90 seconds of idle inactivity, generating a custom UI toast alert.
    \end{itemize}
\end{frame}
